## Research Foundation (NotebookLM)

From LearnLM (Google DeepMind) and University of Queensland research:

1. **Index "Pedagogical Logic" Not Just Content**
   - Index misconception explanations for each distractor (wrong answer)
   - Retrieve WHY something is wrong, not just what's right
   - Separate "student-friendly" vs "teacher-centric" explanations

2. **Atomic Chunking Strategy**
   - Break content into smallest teachable units ("atoms")
   - Wrap chunks in "Activity Definitions" (structured JSON)
   - Segment video into micro-lessons with knowledge checks

## Objective

Set up Pinecone vector database and create content embedding pipeline that indexes course content with pedagogical metadata.

import { GoogleGenerativeAI } from '@google/generative-ai';

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);

export async function embedText(text: string): Promise<number[]> {
  const model = genAI.getGenerativeModel({ model: 'embedding-001' });
  const result = await model.embedContent(text);
  return result.embedding.values;
}

export async function embedBatch(texts: string[]): Promise<number[][]> {
  // Batch embedding for efficiency
  return Promise.all(texts.map(embedText));
}

### Task 3: Content Indexer

**File:** `src/lib/rag/contentIndexer.ts`

Build pipeline to index course content:

```typescript
export async function indexCourseContent(courseId: string): Promise<IndexResult> {
  const course = await getCourse(courseId);
  const chunks: ContentChunk[] = [];

  for (const module of course.modules) {
    for (const lesson of module.lessons) {
      for (const atom of lesson.atoms) {
        // Index main content
        chunks.push(...await chunkAtomContent(atom, courseId, module.id, lesson.id));

        // Index quiz misconceptions
        if (atom.type === 'quiz') {
          chunks.push(...await indexQuizMisconceptions(atom, courseId, module.id, lesson.id));
        }
      }
    }
  }

  // Embed and upsert to Pinecone
  return await upsertChunks(chunks);
}

async function indexQuizMisconceptions(atom, courseId, moduleId, lessonId): Promise<ContentChunk[]> {
  const chunks: ContentChunk[] = [];

  for (const question of atom.questions) {
    for (const option of question.options) {
      if (!option.isCorrect) {
        // Create misconception chunk for each wrong answer
        chunks.push({
          id: `${atom.id}-${question.id}-${option.id}-misconception`,
          courseId, moduleId, lessonId, atomId: atom.id,
          atomType: 'quiz',
          chunkType: 'misconception',
          content: option.misconceptionExplanation || `Why "${option.text}" is incorrect: ...`,
          metadata: {
            questionId: question.id,
            distractorId: option.id,
            misconceptionId: option.misconceptionId,
            studentFriendly: true,
          }
        });
      }
    }
  }

  return chunks;
}
```

### Task 4: Retrieval Service

**File:** `src/lib/rag/retriever.ts`

Create RAG retrieval with pedagogical filtering:

```typescript
export type RetrievalContext = {
  query: string;
  courseId: string;
  lessonId?: string;
  atomId?: string;
  questionId?: string;
  distractorId?: string;  // If student selected wrong answer
  studentAbility?: number; // 0-1 for difficulty filtering
};

export async function retrieveContext(ctx: RetrievalContext): Promise<RetrievedChunk[]> {
  const embedding = await embedText(ctx.query);

  // Build filter based on context
  const filter: Record<string, string> = {
    courseId: ctx.courseId,
  };

  // If we know which wrong answer was selected, prioritize misconception
  if (ctx.distractorId) {
    filter.distractorId = ctx.distractorId;
  }

  const results = await contentIndex.query({
    vector: embedding,
    topK: 5,
    filter,
    includeMetadata: true,
  });

  return results.matches.map(formatChunk);
}

export async function retrieveMisconceptionContext(
  questionId: string,
  distractorId: string
): Promise<RetrievedChunk | null> {
  // Direct lookup for specific misconception
  const results = await contentIndex.query({
    vector: await embedText('misconception explanation'),
    topK: 1,
    filter: { questionId, distractorId, chunkType: 'misconception' },
    includeMetadata: true,
  });

  return results.matches[0] ? formatChunk(results.matches[0]) : null;
}
```

- chunkType: string (content|misconception|hint|example)
- questionId: string (optional)
- distractorId: string (optional)
- misconceptionId: string (optional)
- studentFriendly: boolean
