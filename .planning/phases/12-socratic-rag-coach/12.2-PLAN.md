## Research Foundation (NotebookLM - LearnLM/Google DeepMind)

### LearnLM Prompt Structure
1. **Role Definition**: "Clipped, Socratic style" - concise, question-focused
2. **Negative Constraints**: "Do not give the answer", "One question at a time"
3. **Dynamic Context Injection**: Student ability, misconception mapping
4. **Few-Shot Examples**: Model Socratic responses
5. **Low Temperature**: Deterministic, focused responses

### Three-Tier Scaffolding Model
- **Tier 1**: Metacognitive hints ("What makes you think that?")
- **Tier 2**: Specific questions pointing to error location
- **Tier 3**: Sentence frames / worked examples (last resort)

### Task 1: Socratic Prompt Template

**File:** `src/lib/rag/socraticPrompts.ts`

Create structured prompt system based on LearnLM research:

```typescript
export type StudentContext = {
  name: string;
  predictedAbility: number; // 0-1 scale
  consecutiveWrong: number;
  currentStruggleLevel: 'none' | 'mild' | 'moderate' | 'severe';
  emotionalState?: 'frustrated' | 'confused' | 'engaged' | 'neutral';
};

export type ActivityContext = {
  lessonTitle: string;
  atomType: 'video' | 'reading' | 'quiz' | 'practice';
  questionText?: string;
  studentAnswer?: string;
  correctAnswer?: string;
  misconceptionExplanation?: string;
};

export function buildSocraticSystemPrompt(
  student: StudentContext,
  activity: ActivityContext,
  ragContext: string
): string {
  return `Act as a learning coach named Sage who is currently helping a student named ${student.name} with the activity below in a clipped, Socratic style.

# DIRECTIVES (CRITICAL - NEVER VIOLATE)
- Do NOT give the student the answer directly
- Only ask ONE question at a time
- Keep responses short and focused (2-3 sentences max)
- If the user asks for the answer, ask them to explain their thinking first
- Use plain text only, no markdown or LaTeX
- Be warm but concise - avoid chattiness

# STUDENT CONTEXT
${buildAbilityDirective(student.predictedAbility)}
${buildStruggleDirective(student.currentStruggleLevel)}
${student.emotionalState ? buildEmotionalDirective(student.emotionalState) : ''}

# CURRENT ACTIVITY
Lesson: ${activity.lessonTitle}
Type: ${activity.atomType}
${activity.questionText ? `Question: ${activity.questionText}` : ''}
${activity.studentAnswer ? `Student answered: ${activity.studentAnswer}` : ''}

# MISCONCEPTION CONTEXT (from course materials)
${activity.misconceptionExplanation || 'No specific misconception identified'}

# RELEVANT COURSE CONTENT
${ragContext}

# INTERVENTION HIERARCHY
Follow this order strictly:
1. FIRST: Ask a clarifying/metacognitive question ("What made you think that?", "Can you walk me through your reasoning?")
2. IF STILL STUCK: Point to the specific area of confusion ("Look at what happens when we...")
3. LAST RESORT ONLY: Provide a sentence frame or worked example of a SIMILAR problem

# EXAMPLES OF GOOD SOCRATIC RESPONSES
- "What happens if we consider X first?"
- "Interesting! How did you arrive at that conclusion?"
- "Let's think about this step by step. What's the first thing we need to identify?"
- "You're on the right track! What do you notice about [specific element]?"

# IMPORTANT GUIDELINES
- Never use the word "bot" or "AI"
- If student wants to end, let them go gracefully
- If student knows correct answer, offer to dig deeper (they may be guessing)
- Match your language complexity to student ability level`;
}

function buildAbilityDirective(ability: number): string {
  if (ability < 0.5) {
    return 'The student is predicted to struggle. Use brief, simple language. Break concepts into smaller steps.';
  } else if (ability >= 0.8) {
    return 'The student is predicted to do well. You can use more advanced vocabulary and explore deeper concepts.';
  }
  return 'The student has moderate predicted ability. Balance simplicity with appropriate challenge.';
}

function buildStruggleDirective(level: string): string {
  switch (level) {
    case 'severe':
      return 'ALERT: Student is severely struggling. Be extra supportive. Consider offering a simpler approach or break.';
    case 'moderate':
      return 'Student showing moderate struggle. Provide more scaffolding in your questions.';
    case 'mild':
      return 'Student showing mild struggle. Standard Socratic approach is appropriate.';
    default:
      return '';
  }
}

function buildEmotionalDirective(state: string): string {
  switch (state) {
    case 'frustrated':
      return 'Student appears frustrated. Acknowledge their effort and provide encouragement before questioning.';
    case 'confused':
      return 'Student appears confused. Ask clarifying questions to identify the specific point of confusion.';
    case 'engaged':
      return 'Student appears engaged. Maintain momentum with appropriately challenging questions.';
    default:
      return '';
  }
}
```

### Task 2: Hierarchical Intervention Manager

**File:** `src/lib/rag/interventionManager.ts`

Track and enforce intervention hierarchy:

```typescript
export type InterventionTier = 1 | 2 | 3;

export type InterventionState = {
  currentTier: InterventionTier;
  tier1Attempts: number;
  tier2Attempts: number;
  tier3Used: boolean;
  conceptId: string;
};

export function getInterventionDirective(state: InterventionState): string {
  if (state.tier3Used) {
    // Already provided maximum support - now verify understanding
    return `You've already provided a worked example. Now verify the student understood by asking them to apply it to the current problem.`;
  }

  if (state.currentTier === 1) {
    return `Use TIER 1 intervention: Ask a metacognitive question.
Examples: "What made you choose that answer?", "Can you explain your reasoning?", "What do you already know about this topic?"`;
  }

  if (state.currentTier === 2) {
    return `Use TIER 2 intervention: Point to the specific area of confusion.
Examples: "Look at what happens when X...", "Notice that the question asks about Y, not Z", "Consider the relationship between A and B"`;
  }

  // Tier 3
  return `Use TIER 3 intervention (LAST RESORT): Provide a sentence frame or worked example.
Example: "Let me show you a similar problem: [worked example]. Now, how would you apply this to your question?"
Important: Still don't give the direct answer - show the method on a different example.`;
}

export function advanceTier(state: InterventionState): InterventionState {
  if (state.currentTier === 1 && state.tier1Attempts >= 2) {
    return { ...state, currentTier: 2, tier2Attempts: 0 };
  }
  if (state.currentTier === 2 && state.tier2Attempts >= 2) {
    return { ...state, currentTier: 3 };
  }
  if (state.currentTier === 3) {
    return { ...state, tier3Used: true };
  }

  // Increment attempts at current tier
  if (state.currentTier === 1) {
    return { ...state, tier1Attempts: state.tier1Attempts + 1 };
  }
  return { ...state, tier2Attempts: state.tier2Attempts + 1 };
}

export function createInterventionState(conceptId: string): InterventionState {
  return {
    currentTier: 1,
    tier1Attempts: 0,
    tier2Attempts: 0,
    tier3Used: false,
    conceptId,
  };
}
```

### Task 3: RAG Context Formatter

**File:** `src/lib/rag/contextFormatter.ts`

Format retrieved chunks for injection:

```typescript
export type RetrievedChunk = {
  content: string;
  chunkType: 'content' | 'misconception' | 'hint' | 'example';
  lessonTitle: string;
  score: number;
  metadata: Record<string, unknown>;
};

export function formatRAGContext(
  chunks: RetrievedChunk[],
  maxLength: number = 2000
): string {
  if (chunks.length === 0) {
    return 'No specific course content retrieved for this query.';
  }

  // Prioritize misconception chunks if present
  const misconceptions = chunks.filter(c => c.chunkType === 'misconception');
  const content = chunks.filter(c => c.chunkType === 'content');
  const hints = chunks.filter(c => c.chunkType === 'hint');

  let context = '';

  if (misconceptions.length > 0) {
    context += '## Relevant Misconceptions\n';
    for (const m of misconceptions.slice(0, 2)) {
      context += `From "${m.lessonTitle}": ${m.content}\n\n`;
    }
  }

  if (content.length > 0) {
    context += '## Related Course Content\n';
    for (const c of content.slice(0, 3)) {
      context += `From "${c.lessonTitle}": ${c.content}\n\n`;
    }
  }

  if (hints.length > 0) {
    context += '## Available Hints\n';
    for (const h of hints.slice(0, 2)) {
      context += `- ${h.content}\n`;
    }
  }

  // Truncate if too long
  if (context.length > maxLength) {
    context = context.substring(0, maxLength) + '...[truncated]';
  }

  return context;
}

export function formatMisconceptionContext(
  misconception: RetrievedChunk | null,
  studentAnswer: string
): string {
  if (!misconception) {
    return `The student answered "${studentAnswer}". Guide them to discover why this might not be correct without revealing the answer.`;
  }

  return `The student answered "${studentAnswer}".

Common misconception for this answer:
${misconception.content}

Use this understanding to craft a Socratic question that helps them discover the flaw in their reasoning.`;
}
```

export async function generateSocraticResponse(
  systemPrompt: string,
  interventionDirective: string,
  conversationHistory: Message[]
): Promise<string> {
  const model = genAI.getGenerativeModel({
    model: 'gemini-1.5-flash',
    generationConfig: {
      temperature: 0.3, // Low temperature for focused responses
      maxOutputTokens: 200, // Keep responses concise
      topP: 0.8,
    },
  });

  const chat = model.startChat({
    history: conversationHistory.map(m => ({
      role: m.role === 'user' ? 'user' : 'model',
      parts: [{ text: m.content }],
    })),
    systemInstruction: `${systemPrompt}\n\n# CURRENT INTERVENTION DIRECTIVE\n${interventionDirective}`,
  });

  const result = await chat.sendMessage(conversationHistory[conversationHistory.length - 1].content);
  return result.response.text();
}
