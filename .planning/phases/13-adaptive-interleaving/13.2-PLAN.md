### Task 1: Update Session Builder

**File:** `src/lib/adaptive/sessionBuilder.ts`

Add interleaving to the session building process. Import the interleaver and wire it into the session construction.

```typescript
// Add imports
import {
  getReviewItemsForInterleaving,
  interleaveItems,
  type InterleavedItem,
  DEFAULT_INTERLEAVING_CONFIG,
} from '@/lib/sequencing';

// Update SessionItem type to include review challenge flag
export interface SessionItem {
  type: 'review' | 'learn' | 'practice' | 'quiz' | 'warmup' | 'cooldown';
  itemId: string;
  skillId: string;
  estimatedMinutes: number;
  reason: string;
  order: number;
  isReviewChallenge?: boolean; // NEW: Badge indicator
  metadata?: {
    retrievability?: number;
    similarity?: number;
  };
}

// Update interleaveReviews function to use FSRS-based interleaving
export async function interleaveReviewsWithFSRS(
  items: SessionItem[],
  userId: string,
  courseId: string,
  currentSkillId: string
): Promise<SessionItem[]> {
  // Fetch FSRS data for user
  const fsrsData = await fetchFSRSData(userId);
  const skillMap = await getSkillMapForCourse(courseId);

  // Get review items based on retrievability
  const reviewItems = getReviewItemsForInterleaving(
    fsrsData,
    currentSkillId,
    skillMap,
    DEFAULT_INTERLEAVING_CONFIG
  );

  if (reviewItems.length === 0) {
    // No reviews needed, return original items
    return items;
  }

  // Convert to SessionItem format
  const newItems: InterleavedItem[] = items.map(item => ({
    type: 'new' as const,
    itemId: item.itemId,
    skillId: item.skillId,
    reason: item.reason,
    estimatedMinutes: item.estimatedMinutes,
    isReviewChallenge: false,
  }));

  // Interleave
  const interleaved = interleaveItems(newItems, reviewItems);

  // Convert back to SessionItem with order
  return interleaved.map((item, index) => ({
    type: item.type === 'review' ? 'review' : items.find(i => i.itemId === item.itemId)?.type || 'practice',
    itemId: item.itemId,
    skillId: item.skillId,
    estimatedMinutes: item.estimatedMinutes,
    reason: item.reason,
    order: index,
    isReviewChallenge: item.isReviewChallenge,
    metadata: item.metadata,
  }));
}
```

### Task 3: Create Review Challenge Badge Component

**File:** `src/components/learning/ReviewChallengeBadge.tsx` (NEW)

```typescript
'use client';

import { RefreshCw } from 'lucide-react';
import { motion } from 'framer-motion';
import { cn } from '@/lib/utils';

interface ReviewChallengeBadgeProps {
  className?: string;
  size?: 'sm' | 'md';
}

export function ReviewChallengeBadge({
  className,
  size = 'sm',
}: ReviewChallengeBadgeProps) {
  return (
    <motion.span
      initial={{ scale: 0.8, opacity: 0 }}
      animate={{ scale: 1, opacity: 1 }}
      className={cn(
        'inline-flex items-center gap-1 bg-purple/10 text-purple rounded-full font-medium',
        size === 'sm' && 'px-2 py-0.5 text-xs',
        size === 'md' && 'px-3 py-1 text-sm',
        className
      )}
    >
      <RefreshCw className={cn(
        size === 'sm' && 'w-3 h-3',
        size === 'md' && 'w-4 h-4'
      )} />
      <span>Review Challenge</span>
    </motion.span>
  );
}

export default ReviewChallengeBadge;
```
