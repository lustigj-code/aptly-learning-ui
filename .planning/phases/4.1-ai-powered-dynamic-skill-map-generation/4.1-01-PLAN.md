```typescript
// DynamicSkillMap - extends SkillMap with metadata
interface DynamicSkillMap {
  id: string;              // e.g., "course-1-skillmap"
  courseId: string;        // Reference to course
  version: number;         // For versioning
  status: 'draft' | 'approved' | 'active';
  skills: Record<string, Skill>;  // Same as existing SkillMap
  metadata: {
    generatedAt: Date;
    generatedBy: 'ai' | 'manual';
    model?: string;        // e.g., "gemini-2.0-flash-exp"
    approvedAt?: Date;
    approvedBy?: string;
  };
}

// ParsedCourseContent - input to AI generator
interface ParsedCourseContent {
  courseId: string;
  title: string;
  modules: ParsedModule[];
  totalLessons: number;
  totalAtoms: number;
}

interface ParsedModule {
  id: string;
  title: string;
  objectives: string[];
  lessons: ParsedLesson[];
}

interface ParsedLesson {
  id: string;
  title: string;
  objectives: string[];
  atoms: ParsedAtom[];
}

interface ParsedAtom {
  id: string;
  type: 'video' | 'reading' | 'quiz' | 'practice';
  title: string;
  content: string;  // Text content or transcript
  questions?: ParsedQuestion[];  // For quiz atoms
}

interface ParsedQuestion {
  question: string;
  options: string[];
  correctAnswer: number;
  skill?: string;  // Existing skill mapping if available
}
```

Export all types. Re-export Skill, SkillMap, BKTParameters from bkt.ts for convenience.
