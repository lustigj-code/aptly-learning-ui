```tsx
// Server component
import { getAllSkillMaps, parseAllCourses } from '@/lib/skillmap';

export default async function SkillMapsPage() {
  const skillMaps = await getAllSkillMaps();
  const courses = parseAllCourses();

  // Map courses to their skill map status
  const courseStatus = courses.map(course => {
    const skillMap = skillMaps.find(sm => sm.courseId === course.courseId);
    return {
      courseId: course.courseId,
      title: course.title,
      hasSkillMap: !!skillMap,
      status: skillMap?.status ?? 'none',
      skillCount: skillMap ? Object.keys(skillMap.skills).length : 0,
      generatedAt: skillMap?.metadata.generatedAt,
      version: skillMap?.version ?? 0,
    };
  });

  return (
    <div className="p-6 max-w-4xl mx-auto">
      <h1 className="text-2xl font-bold mb-6">Skill Map Management</h1>

      <div className="bg-white rounded-lg shadow">
        <table className="w-full">
          <thead className="bg-gray-50">
            <tr>
              <th className="px-4 py-3 text-left">Course</th>
              <th className="px-4 py-3 text-left">Status</th>
              <th className="px-4 py-3 text-left">Skills</th>
              <th className="px-4 py-3 text-left">Generated</th>
              <th className="px-4 py-3 text-left">Actions</th>
            </tr>
          </thead>
          <tbody>
            {courseStatus.map(course => (
              <tr key={course.courseId} className="border-t">
                <td className="px-4 py-3">
                  <div className="font-medium">{course.title}</div>
                  <div className="text-sm text-gray-500">{course.courseId}</div>
                </td>
                <td className="px-4 py-3">
                  <StatusBadge status={course.status} />
                </td>
                <td className="px-4 py-3">
                  {course.skillCount > 0 ? `${course.skillCount} skills` : '-'}
                </td>
                <td className="px-4 py-3 text-sm text-gray-500">
                  {course.generatedAt?.toLocaleDateString() ?? '-'}
                </td>
                <td className="px-4 py-3">
                  <ActionButtons course={course} />
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}
```

**Detail page (src/app/admin/skill-maps/[courseId]/page.tsx):**
```tsx
import { getSkillMap, parseCourse } from '@/lib/skillmap';
import { SkillMapEditor } from '@/components/admin/SkillMapEditor';
import { GenerateButton } from './GenerateButton';

export default async function SkillMapDetailPage({
  params,
}: {
  params: { courseId: string };
}) {
  const skillMap = await getSkillMap(params.courseId);
  const course = parseCourse(params.courseId);

  if (!course) {
    return <div>Course not found</div>;
  }

  return (
    <div className="p-6 max-w-6xl mx-auto">
      <div className="flex justify-between items-center mb-6">
        <div>
          <h1 className="text-2xl font-bold">{course.title}</h1>
          <p className="text-gray-500">Skill Map Editor</p>
        </div>
        <div className="flex gap-2">
          {!skillMap && <GenerateButton courseId={params.courseId} />}
          {skillMap?.status === 'draft' && (
            <ApproveButton courseId={params.courseId} />
          )}
        </div>
      </div>

      {skillMap ? (
        <SkillMapEditor skillMap={skillMap} course={course} />
      ) : (
        <EmptyState courseId={params.courseId} />
      )}
    </div>
  );
}
```

**Editor component (src/components/admin/SkillMapEditor.tsx):**
```tsx
'use client';

import { useState } from 'react';
import type { DynamicSkillMap, ParsedCourseContent } from '@/lib/skillmap/types';

export function SkillMapEditor({
  skillMap,
  course,
}: {
  skillMap: DynamicSkillMap;
  course: ParsedCourseContent;
}) {
  const [selectedSkill, setSelectedSkill] = useState<string | null>(null);
  const skills = Object.values(skillMap.skills);

  // Group skills by module
  const skillsByModule = skills.reduce((acc, skill) => {
    const moduleNum = skill.lessonId.split('-')[1] ?? 'other';
    if (!acc[moduleNum]) acc[moduleNum] = [];
    acc[moduleNum].push(skill);
    return acc;
  }, {} as Record<string, typeof skills>);

  return (
    <div className="grid grid-cols-3 gap-6">
      {/* Left: Skill list by module */}
      <div className="col-span-2 bg-white rounded-lg shadow p-4">
        <h2 className="font-semibold mb-4">
          Skills ({skills.length} total)
        </h2>
        {Object.entries(skillsByModule).map(([module, moduleSkills]) => (
          <div key={module} className="mb-4">
            <h3 className="text-sm font-medium text-gray-500 mb-2">
              Module {module}
            </h3>
            <div className="space-y-1">
              {moduleSkills.map(skill => (
                <div
                  key={skill.id}
                  onClick={() => setSelectedSkill(skill.id)}
                  className={`p-2 rounded cursor-pointer ${
                    selectedSkill === skill.id
                      ? 'bg-blue-50 border border-blue-200'
                      : 'hover:bg-gray-50'
                  }`}
                >
                  <div className="font-medium text-sm">{skill.name}</div>
                  <div className="text-xs text-gray-500">
                    Lesson: {skill.lessonId} |
                    Prerequisites: {skill.prerequisites.length}
                  </div>
                </div>
              ))}
            </div>
          </div>
        ))}
      </div>

      {/* Right: Skill detail panel */}
      <div className="bg-white rounded-lg shadow p-4">
        {selectedSkill ? (
          <SkillDetailPanel
            skill={skillMap.skills[selectedSkill]}
            allSkills={skills}
          />
        ) : (
          <div className="text-gray-500 text-center py-8">
            Select a skill to view details
          </div>
        )}
      </div>
    </div>
  );
}

function SkillDetailPanel({ skill, allSkills }) {
  return (
    <div>
      <h3 className="font-semibold mb-4">{skill.name}</h3>
      <div className="space-y-4 text-sm">
        <div>
          <label className="text-gray-500">Skill ID</label>
          <div className="font-mono">{skill.id}</div>
        </div>
        <div>
          <label className="text-gray-500">Lesson</label>
          <div>{skill.lessonId}</div>
        </div>
        <div>
          <label className="text-gray-500">Prerequisites</label>
          <ul className="list-disc list-inside">
            {skill.prerequisites.length > 0
              ? skill.prerequisites.map(prereq => (
                  <li key={prereq}>{allSkills.find(s => s.id === prereq)?.name ?? prereq}</li>
                ))
              : <li className="text-gray-400">None</li>}
          </ul>
        </div>
        <div>
          <label className="text-gray-500">BKT Parameters</label>
          <div className="grid grid-cols-2 gap-2 mt-1">
            <div>pL0: {skill.bktParams.pL0}</div>
            <div>pT: {skill.bktParams.pT}</div>
            <div>pG: {skill.bktParams.pG}</div>
            <div>pS: {skill.bktParams.pS}</div>
          </div>
        </div>
      </div>
    </div>
  );
}
```
