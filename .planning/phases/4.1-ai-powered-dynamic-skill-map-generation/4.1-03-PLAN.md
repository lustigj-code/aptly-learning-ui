1. Add skill map loading function:
```typescript
import { getSkillMap } from '@/lib/skillmap/skillMapStorage';
import { AI_AT_WORK_SKILL_MAP } from '@/data/skillMap'; // Keep as fallback

// In-memory cache
const skillMapCache = new Map<string, SkillMap>();

// Get skill map for course (with caching)
async function getSkillMapForCourse(courseId: string): Promise<SkillMap> {
  // Check cache first
  if (skillMapCache.has(courseId)) {
    return skillMapCache.get(courseId)!;
  }

  // Try Firestore
  const dynamicMap = await getSkillMap(courseId);
  if (dynamicMap && dynamicMap.status === 'active') {
    skillMapCache.set(courseId, { skills: dynamicMap.skills });
    return { skills: dynamicMap.skills };
  }

  // Fallback to hardcoded for AI at Work course
  if (courseId === 'ai-at-work' || courseId === 'course-ai-work') {
    return AI_AT_WORK_SKILL_MAP;
  }

  // Return empty skill map if none found
  console.warn(`No skill map found for course: ${courseId}`);
  return { skills: {} };
}

// Clear cache (for admin/refresh)
export function clearSkillMapCache(courseId?: string): void {
  if (courseId) {
    skillMapCache.delete(courseId);
  } else {
    skillMapCache.clear();
  }
}
```

```typescript
// Re-export all public functions
export * from './types';
export { parseCourse, parseAllCourses } from './courseParser';
export { generateSkillMap, validateSkillMap, mergeSkillMaps } from './skillMapGenerator';
export {
  saveSkillMap,
  getSkillMap,
  getAllSkillMaps,
  updateSkillMapStatus,
  deleteSkillMap,
  hasSkillMap
} from './skillMapStorage';

// Migration helper: Generate and save skill map for a course
export async function generateAndSaveSkillMap(
  courseId: string,
  autoApprove: boolean = false
): Promise<DynamicSkillMap> {
  const courseContent = parseCourse(courseId);
  if (!courseContent) {
    throw new Error(`Course not found: ${courseId}`);
  }

  const skillMap = await generateSkillMap(courseContent);

  if (autoApprove) {
    skillMap.status = 'active';
  }

  await saveSkillMap(skillMap);

  // Clear cache so new map is loaded
  clearSkillMapCache(courseId);

  return skillMap;
}

// Bulk generate for all courses without skill maps
export async function generateMissingSkillMaps(): Promise<{
  generated: string[];
  failed: string[];
}> {
  const courses = parseAllCourses();
  const generated: string[] = [];
  const failed: string[] = [];

  for (const course of courses) {
    const exists = await hasSkillMap(course.courseId);
    if (!exists) {
      try {
        await generateAndSaveSkillMap(course.courseId);
        generated.push(course.courseId);
      } catch (error) {
        console.error(`Failed to generate skill map for ${course.courseId}:`, error);
        failed.push(course.courseId);
      }
    }
  }

  return { generated, failed };
}
```
